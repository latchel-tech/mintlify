### 1 - Get data on Skill Categories and Skills being used in Emergency Jobs

```sql lines wrap
SELECT c.name, i.name, count(*) FROM public.job j
	JOIN public.category c
		ON j.category_id = c.category_id
		AND j.is_emergency = 1
	JOIN public.issue i
		ON j.issue_id = i.issue_id
GROUP BY 1, 2
```

### **2 - 	Emergency Work order information - data dump - including only our internal team**

```sql lines wrap
SELECT u.name, cl.created_at as call_date, cl.duration as call_duration, cl.status as call_status, cl.recording_url, j.name as job_name, concat('https://stage.latchel.com/admin/job/',j.slug) as job_url, * FROM public.call_log cl
JOIN public.user u
ON cl.initiated_by_user_id = u.user_id
AND u.role_id IN (1,6)        LEFT JOIN public.call_log_job clj
ON cl.call_log_id = clj.call_log_id
LEFT JOIN public.job j
ON clj.job_id = j.job_id
and J.IS_EMERGENCY = 1
ORDER BY 2 desc
```

### 4 - Emergency SLAs failures

```sql lines wrap
WITH e as (SELECT l.relation_id as job_id, l.created_at as create_date, * FROM public.log l
WHERE l.type = 'state'
AND l.loggable_id = 25
AND l.relation_type = 'Latchel\Job'
AND DATE_TRUNC('month',l.created_at) >= DATE_Trunc('month',now()-interval '12 months')), t as 
(
SELECT t.relation_id as job_id, min(t.created_at) as tshoot_complete_at 
	FROM e
	JOIN public.log t
		ON e.job_id = t.relation_id
		AND t.relation_type = 'Latchel\Job'
		AND t.type = 'state'
		AND t.created_at > e.create_date
		AND t.loggable_id <> 25
GROUP BY 1)
SELECT em.job_id, em.create_date, t.tshoot_complete_at, u.user_id, u.name, r.name as user_type, EXTRACT(EPOCH FROM t.tshoot_complete_at-em.create_date)/60 as minutes_to_tshootFROM e as em
	JOIN t 
		ON em.job_id = t.job_id
	JOIN public.log l
		ON l.relation_id = t.job_id
		AND l.relation_type = 'Latchel\Job'
		AND l.created_at = t.tshoot_complete_at 
		AND l.type = 'state'
	JOIN public.user u
		ON l.creator_id = u.user_id
	JOIN public.role r
		ON u.role_id = r.role_id
```

### 5 - Data on tasks – last 1 week of data - all users

```sql lines wrap
SELECT u.user_id, u.name, i.day_action_taken, d.tasks_deferred, c.tasks_created, co.tasks_completed, n.notes_created,
l.status_updates, m.conversations_replied, j.wos_created, call.calls_placed
FROM 
(
	SELECT u.user_id, u.name 
	FROM public.user u
	WHERE u.user_type = 'admin'
	AND u.is_Active = 1
	AND u.role_id in (1,6) -- admin/in-house
	AND u.user_id <> 1
	) u
JOIN (
select day_action_taken::date from generate_series(date_trunc('day', now() at time zone 'America/Los_Angeles' - interval '1 week'), 
  now() at time zone 'America/Los_Angeles', '1 day'::interval) day_action_taken) i
  ON 1=1
LEFT JOIN 
	 ( SELECT a.name, a.day_action_taken, count(*) as tasks_deferred
	  FROM (
		SELECT right(rtrim(l.note, '.'),-17) as name, DATE_trunc('day',l.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') as day_action_taken
		FROM
		public.log l
		WHERE l.note LIKE '%Task deferred by %'
		AND DATE_TRUNC('week',l.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') > DATE_trunc('week', now() at time zone 'America/Los_Angeles' - interval '1 week')
		) a 
	 	GROUP BY 1,2) d
	ON u.name = d.name
	AND i.day_action_taken = d.day_action_taken
LEFT JOIN 
(SELECT a.creator_id, a.day_action_taken, count(*) as tasks_created
 FROM (
	SELECT t.creator_id, DATE_trunc('day',t.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') as day_action_taken
	FROM
	public.task t
	WHERE DATE_TRUNC('week',t.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') > DATE_trunc('week', now() at time zone 'America/Los_Angeles' - interval '1 week')
	AND t.creator_id <> 1 ) a
GROUP BY 1, 2) c
	ON u.user_id = c.creator_id
	AND i.day_action_taken = c.day_action_taken
LEFT JOIN 
(SELECT a.user_id, a.day_action_taken, count(*) as tasks_completed
 FROM (
	SELECT t.user_id, DATE_trunc('day',t.completed_at at time zone 'UTC' at time zone 'America/Los_Angeles') as day_action_taken
	FROM
	public.task t
	WHERE DATE_TRUNC('week',t.completed_at at time zone 'UTC' at time zone 'America/Los_Angeles') > DATE_trunc('week', now() at time zone 'America/Los_Angeles' - interval '1 week')
	 ) a
GROUP BY 1, 2) co
	ON u.user_id = co.user_id
	AND i.day_action_taken = co.day_action_taken
LEFT JOIN 
(SELECT a.creator_id, a.day_action_taken, count(*) as notes_created
 FROM (
	SELECT n.creator_id, DATE_trunc('day',n.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') as day_action_taken
	FROM
	public.note n
	WHERE DATE_TRUNC('week',n.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') > DATE_trunc('week', now() at time zone 'America/Los_Angeles' - interval '1 week')
	 ) a
GROUP BY 1, 2) n
	ON u.user_id = n.creator_id
	AND i.day_action_taken = n.day_action_taken
LEFT JOIN 
(SELECT a.creator_id, a.day_action_taken, count(*) as status_updates
 FROM (
	SELECT l.creator_id, DATE_trunc('day',l.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') as day_action_taken
	FROM
	public.log l
	WHERE DATE_TRUNC('week',l.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') > DATE_trunc('week', now() at time zone 'America/Los_Angeles' - interval '2 weeks')
	AND l.loggable_type = 'Latchel\State'
	AND l.creator_id != 1
	 ) a
GROUP BY 1, 2) l
	ON u.user_id = l.creator_id
	AND i.day_action_taken = l.day_action_taken
LEFT JOIN 
(SELECT a.user_id, a.day_action_taken, count(*) as conversations_replied
 FROM (
	SELECT c.user_id, DATE_trunc('day',m.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') as day_action_taken
	FROM
	(
SELECT m.conversation_id, min(m.created_at) as first_msg
		FROM public.message m
		WHERE DATE_TRUNC('week',m.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') > DATE_trunc('week', now() at time zone 'America/Los_Angeles' - interval '1 week')
		GROUP BY 1 ) f
	JOIN public.conversation c
	ON f.conversation_id = c.conversation_id
	JOIN public.message m
	ON f.first_msg = m.created_at
	AND f.conversation_id = m.conversation_id
	AND c.conversable_id = m.messageable_id
	 ) a
GROUP BY 1, 2) m
	ON u.user_id = m.user_id
	AND i.day_action_taken = m.day_action_taken
LEFT JOIN 
(SELECT a.creator_id, a.day_action_taken, count(*) as wos_created
 FROM (
	SELECT j.creator_id, DATE_trunc('day',j.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') as day_action_taken
	FROM
	public.job j
	WHERE DATE_TRUNC('week',j.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') > DATE_trunc('week', now() at time zone 'America/Los_Angeles' - interval '1 week')
	AND j.creator_id <> 1 ) a
GROUP BY 1, 2) j
	ON u.user_id = j.creator_id
	AND i.day_action_taken = j.day_action_taken
LEFT JOIN 
(SELECT a.initiated_by_user_id, a.day_action_taken, count(*) as calls_placed
 FROM (
	SELECT c.initiated_by_user_id, DATE_trunc('day',c.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') as day_action_taken
	FROM
	public.call_log c
	WHERE DATE_TRUNC('week',c.created_at at time zone 'UTC' at time zone 'America/Los_Angeles') > DATE_trunc('week', now() at time zone 'America/Los_Angeles' - interval '1 week')
	) a
GROUP BY 1, 2) call
	ON u.user_id = call.initiated_by_user_id
	AND i.day_action_taken = call.day_action_taken
```

### 10 - Emergency Call Back Times

```sql lines wrap
WITH e as (SELECT l.relation_id as job_id, l.created_at FROM public.log l
WHERE l.type = 'state'
AND l.loggable_id = 25
AND l.relation_type = 'Latchel\Job'
AND DATE_TRUNC('month',l.created_at) >= DATE_Trunc('month',now()-interval '12 months'))
SELECT e.job_id, e.created_at at time zone 'utc' at time zone 'America/Los_Angeles' as created_at, c.first_call at time zone 'utc' at time zone 'America/Los_Angeles' as first_call, cl.initiated_by_user_id, u.name, r.name as user_type, GREATEST(EXTRACT(EPOCH FROM c.first_call-e.created_at)/60,0) as minutes_to_call
FROM (
	SELECT e.job_id, min(cl.created_at) as first_call FROM
		e
		JOIN public.call_log_job clj
			ON e.job_id = clj.job_id
		JOIN public.call_log cl
			ON clj.call_log_id = cl.call_log_id
			AND (cl.created_at + cl.duration * interval '1 second' ) >= e.created_at
	GROUP BY 1) c
	JOIN e
		ON c.job_id = e.job_id
	JOIN public.call_log cl
		ON cl.created_at = c.first_call
	JOIN public.call_log_job clj
		ON clj.job_id = c.job_id
		AND clj.call_log_id = cl.call_log_id
	JOIN public.user u
		ON cl.initiated_by_user_id = u.user_id
	JOIN public.role r
		ON u.role_id = r.role_id
```

### 11 - Source of **Work Orders** - grouped by **day** - form used and creator

```sql lines wrap
WITH a as (SELECT * FROM generate_series(date_trunc('day',now()-interval '1 months'),
                              date_trunc('day',now()), '1 day') as date)
SELECT a.date, c.name, j.source, cr.user_type, count(*) 
from a
JOIN public.job j
	ON date_trunc('day',j.created_at) = a.date
	JOIN public.user pm
		ON j.pm_id = pm.user_id
	JOIN public.company c
		ON pm.company_id = c.company_id
	JOIN public.user cr
		ON j.creator_id = cr.user_id
GROUP BY 1,2,3,4
ORDER BY 1,2,3,4
```

### 12 - Source of **Work Orders** - grouped by **hour** - form used and creator ---\> **UTC time zone**

```sql lines wrap
WITH a as (SELECT * FROM generate_series(date_trunc('day',now()-interval '1 months'),
                              date_trunc('hour',now()), '1 hour') as hour)
SELECT a.hour, c.name, j.source, cr.user_type, count(*)
```

### 13 - Number of units per active PM on the day the query is run

- PM account creation date

```sql lines wrap
Select distinct (c.name), c.created_at, count (distinct pr.property_id)  from company c
join public.user pm
on c.company_id=pm.company_id
join property pr on pr.pm_id=pm.user_id
where c.company_type = 'pm'
and c.is_active = '1'
and pr.is_active = '1'
```

### 52 - Web Chat Missed SLA Intervals (BigQuery query)

```sql lines wrap
WITH f AS (
  SELECT
    c.conversation_id,
    DATETIME(CAST(c.created_at AS TIMESTAMP), 'America/Los_Angeles') AS created_at_pacific
  FROM `rt_sms_web_public.conversation` AS c
  WHERE
    c.type = 'web'
      AND DATE(DATETIME(CAST(c.created_at AS TIMESTAMP), 'America/Los_Angeles')) BETWEEN PARSE_DATE('%Y%m%d', '20251026') AND PARSE_DATE('%Y%m%d', '20251101') --<------------------- CHANGE DATE HERE
),

pmc AS (
  SELECT
    f.conversation_id,
    f.created_at_pacific,
    COALESCE(u.user_type, 'unknown') AS contact_type
  FROM f
  INNER JOIN `rt_sms_web_public.conversation_member` AS cm
    ON f.conversation_id = cm.conversation_id
  LEFT JOIN `emergency_dataset_public.user` AS u
    ON cm.user_id = u.user_id
),

raw AS (
  SELECT 
    pmc.conversation_id,
    pmc.created_at_pacific,
    u.user_id,
    u.name AS latchel_user,
    pmc.contact_type,
    TIMESTAMP_DIFF(
      DATETIME(CAST(MIN(m.created_at) AS TIMESTAMP), 'America/Los_Angeles'),
      pmc.created_at_pacific,
      MINUTE
    ) AS reply_minutes
  FROM pmc
  INNER JOIN `rt_sms_web_public.message` AS m 
    ON pmc.conversation_id = m.conversation_id
  INNER JOIN `rt_sms_web_public.conversation_member` AS cm 
    ON m.conversation_member_id = cm.conversation_member_id
  INNER JOIN `emergency_dataset_public.user` AS u 
    ON cm.user_id = u.user_id
    AND u.user_type = 'admin'
    AND u.user_id <> 1
GROUP BY 1,2,3,4,5
)
SELECT
  DATETIME_TRUNC(DATETIME(raw.created_at_pacific), QUARTER) AS quarter,
  DATETIME_TRUNC(DATETIME(raw.created_at_pacific), MONTH) AS month,
  EXTRACT(ISOWEEK FROM DATETIME(raw.created_at_pacific)) AS week,
  DATETIME_TRUNC(DATETIME(raw.created_at_pacific), DAY) AS day,
  CAST(EXTRACT(HOUR FROM DATETIME(raw.created_at_pacific)) AS FLOAT64) AS hour,
  APPROX_QUANTILES(raw.reply_minutes, 100)[OFFSET(50)] AS p50,
  APPROX_QUANTILES(raw.reply_minutes, 100)[OFFSET(75)] AS p75,
  APPROX_QUANTILES(raw.reply_minutes, 100)[OFFSET(90)] AS p90,
  COUNT(DISTINCT raw.conversation_id) AS conv_count
FROM 
  raw
GROUP BY 
  quarter,
  month,
  week,
  day,
  hour
ORDER BY 
  day DESC,
  hour ASC;
```
