---
title: "Job Created & Updated"
description: "The following data is sent when a Job is created or updated in Latchel. (See Payload Format, above)"
---

## Request Body schema:  application/json

<ParamField body="job_id" type="integer">
  Unique ID for Job
</ParamField>

<ParamField body="slug" type="string">
  Unique 5 digit string used to identify Job
</ParamField>

<ParamField body="source" type="string">
  Source of the Job, such as Tenant Web Form, Appfolio Plugin, Internal Web Form, Tenant Call, or API Request
</ParamField>

<ParamField body="created_at" type="string">
  string \<date-time\>
</ParamField>

<ParamField body="updated_at" type="string">
  string \<date-time\>
</ParamField>

<ParamField body="state_id" type="integer">
  ID of current Job State
</ParamField>

<ParamField body="state" type="object">
  State object (schemas)
</ParamField>

<ParamField body="state_updated_at" type="string">
  string \<date-time\>

  Date and time of the last state change
</ParamField>

<ParamField body="scheduled_start" default="null" type="string">
  Date and time the work is scheduled to start. This, combined with scheduled\_end, creates the arrival window for the Vendor.
</ParamField>

<ParamField body="scheduled_end" default="null" type="string">
  Date and time the work is scheduled to end. This, combined with scheduled\_start, creates the arrival window for the Vendor.
</ParamField>

<ParamField body="is_urgent" default="0" type="integer">
  1 or 0 representing if the Job is Urgent severity
</ParamField>

<ParamField body="is_emergency" default="0" type="integer">
  1 or 0 representing if the Job is Emergency severity
</ParamField>

<ParamField body="job_rating" default="null" type="integer">
  Rating (by Tenant) of the service received
</ParamField>

<ParamField body="job_rated_at" default="null" type="string">
  Date and time the Tenant left a Job rating
</ParamField>

<ParamField body="rating_comments" default="null" type="string">
  Comments added by Tenant with Job rating
</ParamField>

<ParamField body="order_number" type="string">
  Mutable identifier for the Job
</ParamField>

<ParamField body="name" type="string">
  Human readable name (title)
</ParamField>

<ParamField body="description" type="string">
  Long description of the problem to be solved
</ParamField>

<ParamField body="vendor_description" type="string">
  Description of problem and work to performed, specifically for the Vendor
</ParamField>

<ParamField body="vendor_rating" type="integer">
  Vendor rating given by the tenant for this job
</ParamField>

<ParamField body="access_instructions" default="null" type="string">
  Instructions for entering the Property if the Tenant is not present. Access instructions are required to set enter\_permission = 1 (see enter\_permission for more information).
</ParamField>

<ParamField body="max_cost" default="null" type="float">
  Cost Vendor not to exceed for the Job
</ParamField>

<ParamField body="estimate" default="null" type="float">
  Estimate amount provided by the vendor
</ParamField>

<ParamField body="estimate_note" default="null" type="string">
  Note provided by the vendor explaining the estimate amount
</ParamField>

<ParamField body="enter_permission" default="0" type="integer">
  1 or 0 representing if the Vendor has permission to enter the Property if the Tenant is not present. To set enter\_permission = 1, access instructions must be included with the request, OR default\_access\_instructions must be set for the Job's property (an \[INHERITABLE\] field).
</ParamField>

<ParamField body="is_vacant" default="0" type="integer">
  1 or 0 representing if the Property is vacant
</ParamField>

<ParamField body="is_mandatory" default="0" type="integer">
  1 or 0 representing if the Job is mandatory (i.e., the Tenant cannot cancel the Job)
</ParamField>

<ParamField body="is_tenant_caused" default="0" type="integer">
  1 or 0 representing if the issue needing repairs was caused by the Tenant
</ParamField>

<ParamField body="is_resident_responsible" default="0" type="integer">
  1 or 0 representing if the Tenant is responsible for payment of the Job's Invoice
</ParamField>

<ParamField body="is_archived" default="0" type="integer">
  1 or 0 representing if the Job is archived. While archived, the Job will not show up in search results in the UI
</ParamField>

<ParamField body="managed_by" default="latchel" type="string">
  The party currently managing the Job ("latchel" or "pm")
</ParamField>

<ParamField body="vendor_locked_in" default="0" type="integer">
  1 or 0 representing if the Vendor currently assigned to the Job can be changed
</ParamField>

<ParamField body="tenant_id" default="null" type="integer">
  ID referencing Tenant the Job is for
</ParamField>

<ParamField body="pm_id" type="integer">
  ID referencing the PropertyManager in charge of Job (defaults to primary User in Property Manager Company)
</ParamField>

<ParamField body="ref_job_id" default="null" type="string">
  Reference ID for the Job; use this field to reference the ID of a Job within a 3rd party system (with which you may be integrating).
</ParamField>

<ParamField body="ref_service_request_id" default="null" type="string">
  Additional reference ID for the Job used by integrations that implement the concept of service requests
</ParamField>

<ParamField body="severity" default="normal" type="string">
  String representing Job severity. Enum: "emergency" "urgent" "normal"
</ParamField>

<ParamField body="issue_id" default="null" type="integer">
  ID referencing Issue for Job
</ParamField>

<ParamField body="vendor_id" default="null" type="integer">
  ID referencing Vendor assigned to Job
</ParamField>

<ParamField body="property_id" type="integer">
  ID referencing the Property the Job is for
</ParamField>

## Response

<ResponseField name="200">
  Return a 200 status to indicate that the data was received successfully
</ResponseField>
